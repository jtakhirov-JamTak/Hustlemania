/**
 * The thin state: one finished sprint, no row clearing n≥3. No row quotes a sprint,
 * because none qualifies — the `from {sprint}` attribution exists only to say where
 * the bars came from, so it is absent exactly when there are no bars.
 */
const thin = (rows: InsightRow[], has: string): InsightRow[] =>
  rows.slice(0, 2).map((r) => ({
    ...r,
    tail: "Not enough data",
    tailTone: undefined,
    sub: r.sub.replace(/ · \d+ sprints?( · from .*)?$/, " · 1 sprint").replace(/Present on \d+ of \d+/, "Present on 2 of 6").replace(/Used on \d+ of \d+/, "Used on 2 of 6"),
    bars: r.bars.map((b) => ({ ...b, value: "—", pct: 0 })),
    note: `${has} · In 1 sprint · no single sprint has enough days yet`,
    warn: true,
  }));

export default async function AcrossMockup({ searchParams }: { searchParams: Promise<Record<string, string | string[] | undefined>> }) {
  const sp = await searchParams;
  const theme = themeFromCookie((await cookies()).get(THEME_COOKIE)?.value);
  const view: View = VIEWS.includes(sp.view as View) ? (sp.view as View) : "full";
  const scope = view === "chip-empty" ? "health" : typeof sp.scope === "string" ? sp.scope : "all";

  return (
    <div className="page" data-mock>
      <header className="app-header">
        <div className="brand">
          <span className="brand-dot" />
          <span className="heading brand-name">Hustlemania</span>
        </div>
        <nav className="tabs" aria-label="Sections">
          <a className="tab" href="#">
            Sprints
          </a>
          <a className="tab" href="#">
            Vision
          </a>
          <a className="tab tab-on" href="#">
            Insights
          </a>
        </nav>
        <div className="header-spacer" />
        <ThemeToggle theme={theme} />
        <div className="header-user">
          <span className="header-email hide-narrow">you@example.com</span>
          <span className="link-quiet nowrap">Sign out</span>
        </div>
      </header>

      <div className="shell" data-shell>
        <Sidebar empty={view === "none"} />
        <main className="main">
          <div className="workspace">
            <nav className="mk-views" aria-label="Mockup views">
              {VIEWS.map((v) => (
                <a key={v} href={`/mockup/across?view=${v}`} aria-current={v === view ? "page" : undefined}>
                  {v}
                </a>
              ))}
            </nav>
            {view === "none" ? <Placeholder /> : <Across view={view} scope={scope} />}
          </div>
        </main>
      </div>
    </div>
  );
}

function Placeholder() {
  return (
    <div className="card card-page">
      <span className="tag tag-accent">Across sprints</span>
      <h1 className="heading page-title mt-12">Patterns arrive after your first sprint</h1>
      <p className="lede lede-narrow">
        Each finished sprint gets a postmortem on the left. Once a few of them exist, this page compares them: which impediments line up with
        under-target days, which cues line up with good ones.
      </p>
    </div>
  );
}

function Across({ view, scope }: { view: View; scope: string }) {
  const bare = view === "chip-empty";
  const isThin = view === "thin";

  const stats = bare ? "No finished Health sprint" : isThin ? "1 sprint · 6 closed days · 2 on target (33%)" : "3 sprints · 34 closed days · 19 on target (56%)";
  const kicker = bare ? "Evidence · nothing yet" : isThin ? "Evidence · Sep 1 → today" : "Evidence · Jul 1 → today";
  const coverage = bare ? "No closed days" : isThin ? "Logged 6 of 6 closed days" : "Logged 31 of 34 closed days · 3 unsure";
  const occ = bare ? "No occurrences" : isThin ? "2 occurrences · 2 answered" : "9 occurrences · 7 answered";
  const resp = bare ? "No responses" : isThin ? "2 responses · 1 answered" : "7 responses · 6 answered";

  const emptyImp = bare ? "No finished Health sprint yet. Comparisons appear once one ends." : null;
  const emptyFt = bare ? "No finished Health sprint yet." : null;

  return (
    <>
      <div className="lg-head">
        <div>
          <div className="lg-kicker">{kicker}</div>
          <h1 className="heading page-title mt-6">What actually works for you</h1>
        </div>
        <div>
          <div className="lg-stats">{stats}</div>
          <div className="lg-caveat">Comparisons need 3 days on each side · association, not cause</div>
        </div>
      </div>

      <div className="lg-scopes" role="group" aria-label="Scope">
        {SCOPES.map(([key, label]) => (
          <a key={key} className={`chip ${scope === key ? "chip-on" : ""}`} href={`/mockup/across?view=${view}&scope=${key}`}>
            {label}
          </a>
        ))}
      </div>

      <div className="pm-cards">
        <InsightCard
          tone="impediment"
          title="IMPEDIMENT IMPACT"
          coverage={coverage}
          question="Median daily attainment on days an obstacle was present vs absent."
          empty={emptyImp}
          rows={bare ? [] : isThin ? thin(IMPACT, "Needs 3 days with and 3 without · has 2 and 4") : IMPACT}
          testId="card-impact"
        />
        <InsightCard
          tone="response"
          title="RESPONSE FOLLOW-THROUGH"
          coverage={occ}
          question="When the highest impediment showed up, did the WHEN → THEN response run?"
          empty={emptyFt}
          rows={bare ? [] : isThin ? thin(FOLLOW, "Needs 3 answered occurrences · has 2") : FOLLOW}
          testId="card-followthrough"
        />
        <InsightCard
          tone="response"
          title="RESPONSE RECOVERY"
          coverage={resp}
          question="After the response ran, was the recovery criterion met?"
          empty={emptyFt}
          rows={bare ? [] : isThin ? thin(RECOVERY, "Needs 3 answered responses · has 1") : RECOVERY}
          testId="card-recovery"
        />
        <InsightCard
          tone="cue"
          title="CUE USEFULNESS"
          coverage={coverage}
          question="Median daily attainment on days a cue was used vs not used."
          empty={emptyImp}
          rows={bare ? [] : isThin ? thin(CUES, "Needs 3 days with and 3 without · has 2 and 4") : CUES}
          testId="card-cues"
        />
      </div>

      <div className="lg-foot">
        <section className="card lg-suggest">
          <div className="lg-title">Suggested kit for the next sprint</div>
          <p className="lg-body">{bare ? SUGGEST_NONE : isThin ? SUGGEST_THIN : SUGGEST_FULL}</p>
        </section>
        <section className="card lg-read">
          <div className="lg-title">How to read this</div>
          <p className="lg-body">{HOW_TO_READ}</p>
        </section>
      </div>
    </>
  );
}

function Sidebar({ empty }: { empty: boolean }) {
  const rows = [
    { label: "Land three retainer clients", meta: "Reviewed Sep 4, 2026", accent: false, sub: "Wealth · Aug 12 → 25", result: ["Met", "met", "112% of goal · completed early"] },
    { label: "Ship the strength base", meta: "Needs review", accent: true, sub: "Health · Jul 29 → Aug 11", result: ["Under", "under", "78% of goal · sprint ended"] },
    { label: "Two calls a week with Dad", meta: "Reviewed Jul 16, 2026", accent: false, sub: "Relationships · Jul 1 → 14", result: ["Met", "met", "104% of goal · sprint complete"] },
  ];

  return (
    <aside data-sidebar>
      <div className="label-muted side-title">Reviews</div>
      {empty ? (
        <div className="side-sub side-empty">Postmortems open here when a sprint ends.</div>
      ) : (
        rows.map((r) => (
          <a key={r.label} className="side-link" href="#">
            <span className="side-link-head">
              <span className="side-label">{r.label}</span>
              <span className={`side-meta ${r.accent ? "side-meta-accent" : ""}`}>{r.meta}</span>
            </span>
            <span className="side-sub">{r.sub}</span>

              <strong data-state={r.result[1]}>{r.result[0]}</strong> · {r.result[2]}
            </span>
          </a>

        <span className="side-link-head">
          <span className="side-label">What works for you</span>
        </span>
      </a>
    </aside>
  );
}
