type Row = {
  name: string;
  tag?: string;
  tail: string;
  tailTone?: "good" | "bad";
  sub: string;
  bars: { value: string; label: string; pct: number; tone?: "good" | "bad" }[];
  note: string | null;
  warn?: boolean;
};

function InsightCard({ tone, title, coverage, question, empty, rows }: { tone: "impediment" | "response" | "cue"; title: string; coverage: string; question: string; empty: string | null; rows: Row[] }) {
  return (
    <div className="card ic">
      <div className="ic-head">
        <span className="ic-kicker" data-tone={tone}>
          {title}
        </span>
        <span className="ic-coverage">{coverage}</span>
      </div>
      <div className="ic-question">{question}</div>
      {empty ? <div className="ic-empty">{empty}</div> : null}
      {empty
        ? null
        : rows.map((r) => (
            <div key={r.name} className="ic-row">
              <div className="ic-row-head">
                <span className="ic-name">
                  {r.name}
                  {r.tag ? <span className="ic-tag">{r.tag}</span> : null}
                </span>
                <span className="ic-tail" data-tone={r.tailTone}>
                  {r.tail}
                </span>
              </div>
              <div className="ic-sub">{r.sub}</div>
              {r.bars.map((b) => (
                <div key={b.label} className="ic-bar">
                  <div className="ic-track">
                    <span className="ic-fill" data-tone={b.tone} style={{ width: `${b.pct}%` }} />
                  </div>
                  <span className="ic-bar-label">
                    <strong>{b.value}</strong> {b.label}
                  </span>
                </div>
              ))}
              {r.note ? (
                <div className="ic-note" data-warn={r.warn ? "true" : "false"}>
                  {r.note}
                </div>
              ) : null}
            </div>
          ))}
    </div>
  );
}

function Decision({ name, sub, opts, on, editable }: { name: string; sub: string; opts: string[]; on: number; editable: boolean }) {
  return (
    <div className="pm-dec">
      <span>
        <span className="pm-dec-name">{name}</span> <span className="pm-dec-sub">{sub}</span>
      </span>
      <span className="pm-dec-opts" role="group" aria-label={`Carry forward ${name}`}>
        {opts.map((o, i) => (
          <button key={o} type="button" className={`chip chip-mini ${o === "Drop" ? "chip-drop" : ""} ${i === on ? "chip-on" : ""}`} aria-pressed={i === on} disabled={!editable}>
            {o}
          </button>
        ))}
      </span>
    </div>
  );
}

function Day({ n, date, actual, target, state, tasks }: { n: number; date: string; actual: string; target: string; state: "met" | "under" | "missed"; tasks: [string, boolean][] }) {
  return (
    <div className="pm-day">
      <div className="pm-day-label">
        Day {n}
        <br />
        {date}
      </div>
      <div>
        <span className="pm-day-actual" data-state={state}>
          {actual}
        </span>{" "}
        <span className="pm-day-label">{state === "missed" ? `· target ${target}` : `of ${target}`}</span>
        {tasks.length ? (
          <ul className="pm-day-tasks">
            {tasks.map(([t, done]) => (
              <li key={t} data-done={done ? "true" : "false"}>
                {t}
              </li>
            ))}
          </ul>
        ) : null}
      </div>
    </div>
  );
}
